\section{电磁感应 III}

\subsection{作业}

\begin{problem}
	习题 8.2
	\begin{solution}
		\begin{enumerate}
			\item[\textbf{1)}] 位移电流密度
			$$
			\dv{\vect{D}}{t} = \dv{t} \ab(\varepsilon_0 E_0 \sin \omega t) = \varepsilon_0 E_0 \omega \cos \omega t
			$$
			\item[\textbf{2)}] $t = 2.0 \times 10^{-5}\,\mathrm{s}$ 时，代入数据得到
			$$
			\vect{D} = 2.0 \times 10^{-3} \,\mathrm{C / m^2}
			$$
			根据安培环路定理得到
			$$
			2 \pi r B = D \cdot \pi r^2 \implies B = \frac{r D}{2} = 2.0 \times 10^{-5} \,\mathrm{T}
			$$
		\end{enumerate}
	\end{solution}
\end{problem}
