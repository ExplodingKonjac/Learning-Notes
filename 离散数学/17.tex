\section{关系 III \& 函数}

\subsection{序关系 continue}

\begin{definition}[最大值和最小值]
    设 $\langle A, \preceq \rangle$ 是偏序结构，且 $S \subseteq A$ 那么称
    \begin{itemize}
        \item $b$ 是 $S$ 的\textbf{最大值} 当且仅当 $b \in S \land \forall\,x \in S: x \preceq b$；
        \item $b$ 是 $S$ 的\textbf{最小值} 当且仅当 $b \in S \land \forall\,x \in S: b \preceq x$；
    \end{itemize}
\end{definition}

\begin{definition}[上界和下界]
    设 $\langle A, \preceq \rangle$ 是偏序结构，且 $S \subseteq A$ 那么称
    \begin{itemize}
        \item $b$ 是 $S$ 的\textbf{上界} 当且仅当 $b \in A \land \forall\,x \in S: x \preceq b$；
        \item $b$ 是 $S$ 的\textbf{下界} 当且仅当 $b \in A \land \forall\,x \in S: b \preceq x$；
        \item $b$ 是 $S$ 的\textbf{上确界} 当且仅当它是所有上界的最小值；
        \item $b$ 是 $S$ 的\textbf{下确界} 当且仅当它是所有下界的最大值；
    \end{itemize}
\end{definition}

\begin{definition}[全序结构]
    设有偏序结构 $\langle A, \preceq \rangle$，若 $\preceq$ 还满足
    $$
    \forall\,x, y \in A: x \preceq y \lor y \preceq x
    $$
    那么称 $\preceq$ 为\textbf{全序关系}，$\langle A, \preceq \rangle$ 是\textbf{全序结构}。
\end{definition}

\begin{definition}[良序结构]
    设有偏序结构 $\langle A, \preceq \rangle$，若 $A$ 的每个非空子集都存在最小元，那么称 $\preceq$ 是\textbf{良序关系}，$\langle A, \preceq \rangle$ 是\textbf{良序结构}。
\end{definition}

良序关系一定是全序关系，但是反之未必成立。比如 $\langle \ZZ, \le \rangle$ 是全序关系，但是 $\ZZ$ 本身就不存在最小元。

\subsection{等价关系}

\begin{definition}[等价关系]
    称关系 $R$ 是一个\textbf{等价关系}当且仅当它是自反的、对称的、传递的。
\end{definition}

\begin{definition}[等价类]
    设 $R$ 是 $A$ 上的等价关系。

    对于每个 $x \in A$，称 $\{y \in A: \langle x, y \rangle \in R\}$ 为 $x$ 关于 $R$ 的\textbf{等价类}，简称为 $x$ 的等价类，记做 $[x]_R$。
    
    称集合 $\{[x]_R: x \in A\}$ 为 $A$ 关于 $R$ 的\textbf{商集}，记做 $A / R$；称 $|A / R|$ 为 $R$ 的\textbf{秩}。
\end{definition}

\subsection{函数}

函数可以视为一类特殊的二元关系，下面给出其定义。

\begin{definition}[部分函数]
    如果从集合 $X$ 到 $Y$ 的二元关系 $f$ 是\textbf{单值}的，即
    $$
    \forall\,x \in X,\ y_1, y_2 \in Y: \langle x, y_1 \rangle \in f \land \langle x, y_2 \rangle \in f \implies y_1 = y_2
    $$
    那么称 $f$ 是 $X \to Y$ 的\textbf{部分函数}。
\end{definition}

\begin{definition}[部分函数]
    如果从集合 $X$ 到 $Y$ 的二元关系 $f$ 是\textbf{单值}的，且
    $$
    \forall\,x \in X: \exists\,y \in Y: \langle x, y \rangle \in f
    $$
    那么称 $f$ 是 $X \to Y$ 的\textbf{函数}。
\end{definition}

若 $f$ 是部分函数，且 $\langle x, y \rangle \in f$，那么称

\begin{itemize}
    \item $y$ 是 $x$ 的\textbf{像点}；
    \item $x$ 是 $y$ 的一个\textbf{源像点}。
\end{itemize}

进一步地：

\begin{definition}
    设 $f$ 是 $X \to Y$ 的部分函数，$A \subseteq X, B \subseteq Y$，那么记
    \begin{itemize}
        \item $f[A] = \{y \in Y: \exists\,x \in A: y = f(x)\}$；
        \item $f^{-1}[B] = \{x \in X: \exists\,y \in B: y = f(x)\}$。
    \end{itemize}
    称 $f[A]$ 为 $A$ 在 $f$ 下的\textbf{像}，$f^{-1}(B)$ 为 $B$ 在 $f$ 下的\textbf{源像}。
\end{definition}

\begin{definition}[函数的限制和沿拓]
    设 $f: X \to Y,\ A \subseteq X$，那么
    \begin{itemize}
        \item 称 $f \cap (A \times Y)$ 为 $f$ 在 $A$ 上的\textbf{限制}，记做 $f \mid_A$。
        \item 称 $f$ 为 $f \mid_A$ 的\textbf{沿拓}。
    \end{itemize}
\end{definition}
